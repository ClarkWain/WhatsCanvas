cmake_minimum_required(VERSION 3.16)
project(Xiangqi LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(MSVC)
    add_compile_options(/utf-8 /wd4819)
endif()

find_package(Threads REQUIRED)
add_library(XiangqiCore Rules.cpp AI.cpp Game.cpp Motion.cpp)
target_include_directories(XiangqiCore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(XiangqiCore PUBLIC Threads::Threads)
add_executable(XiangqiTests tests/Tests.cpp)
target_link_libraries(XiangqiTests PRIVATE XiangqiCore)
if(BUILD_TESTING)
    add_test(NAME XiangqiRulesAndPlay COMMAND XiangqiTests)
    set_tests_properties(XiangqiRulesAndPlay PROPERTIES TIMEOUT 90)
endif()

option(XIANGQI_BUILD_DESKTOP "Build the GLFW/WhatsCanvas desktop game" ON)
if(XIANGQI_BUILD_DESKTOP)
    set(PROJECT_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../..)
    include(${PROJECT_ROOT}/cmake/WhatsCanvasOpenGL.cmake)
    whatscanvas_add_common_dependencies(${PROJECT_ROOT})
    whatscanvas_add_opengl_library(WhatsCanvasXiangqiOpenGL ${PROJECT_ROOT})
    add_executable(Xiangqi Main.cpp Renderer.cpp AnimationTests.cpp)
    target_link_libraries(Xiangqi PRIVATE XiangqiCore WhatsCanvasXiangqiOpenGL)
    whatscanvas_link_gl_app(Xiangqi "${PROJECT_ROOT}")
    if(BUILD_TESTING)
        add_test(NAME XiangqiDesktopSmoke COMMAND Xiangqi --smoke-test)
        set_tests_properties(XiangqiDesktopSmoke PROPERTIES TIMEOUT 60 LABELS "desktop;opengl")
        add_test(NAME XiangqiAnimationSmoke COMMAND Xiangqi --animation-test)
        set_tests_properties(XiangqiAnimationSmoke PROPERTIES TIMEOUT 60 LABELS "desktop;opengl;animation")
    endif()
endif()
