cmake_minimum_required(VERSION 3.16)
project(Chess LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(MSVC)
    add_compile_options(/utf-8 /wd4819)
endif()

find_package(Threads REQUIRED)
add_library(ChessCore Rules.cpp AI.cpp Game.cpp Motion.cpp)
target_include_directories(ChessCore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ChessCore PUBLIC Threads::Threads)
add_executable(ChessTests tests/Tests.cpp)
target_link_libraries(ChessTests PRIVATE ChessCore)
if(BUILD_TESTING)
    add_test(NAME ChessRulesAndPlay COMMAND ChessTests)
    set_tests_properties(ChessRulesAndPlay PROPERTIES TIMEOUT 90)
endif()

option(CHESS_BUILD_DESKTOP "Build the GLFW/WhatsCanvas desktop game" ON)
if(CHESS_BUILD_DESKTOP)
    set(PROJECT_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../..)
    include(${PROJECT_ROOT}/cmake/WhatsCanvasOpenGL.cmake)
    whatscanvas_add_common_dependencies(${PROJECT_ROOT})
    whatscanvas_add_opengl_library(WhatsCanvasChessOpenGL ${PROJECT_ROOT})
    add_executable(Chess Main.cpp Renderer.cpp PieceArt.cpp AnimationTests.cpp)
    target_link_libraries(Chess PRIVATE ChessCore WhatsCanvasChessOpenGL)
    whatscanvas_link_gl_app(Chess "${PROJECT_ROOT}")
    if(BUILD_TESTING)
        add_test(NAME ChessDesktopSmoke COMMAND Chess --smoke-test)
        set_tests_properties(ChessDesktopSmoke PROPERTIES TIMEOUT 60 LABELS "desktop;opengl")
        add_test(NAME ChessAnimationSmoke COMMAND Chess --animation-test)
        set_tests_properties(ChessAnimationSmoke PROPERTIES TIMEOUT 60 LABELS "desktop;opengl;animation")
    endif()
endif()
